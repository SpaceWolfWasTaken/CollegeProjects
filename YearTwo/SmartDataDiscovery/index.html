def drop_columns(df:DataFrame, *args):
    '''
    Takes a pandas dataframe object and an arbitary number of column names as argument.
    Loops and removes the columns, printing success or failure. The columns are removed inplace.
    '''
    if len(args) == 0:
        #return if no columns have been specified.
        print("No columns specified.")
        return
    for arg in args: #loop through the arguments
        if arg in df.columns: #check if the given argument is a column.
            df.drop(columns=arg, inplace=True) #remove the column.
            print(f"Column '{arg}' dropped.")
        else:
            print(f"Column '{arg}' does not exist in the dataframe.")

def remove_nan(df:DataFrame):
    '''
    Checks if NaN values exist, removing the rows if they do.
    '''
    if not df.isna().values.any(): #if no NaN values exist
        print("No NaN values exist.")
        return
    df.dropna(inplace=True, axis=ROW) #remove the NaN values, since they do exist if the code reaches here

def check_duplicates(df:DataFrame) -> bool:
    '''
    Returns True if duplicates exist. Returns False if duplicates do not exist.
    '''
    total_duplicates = df.duplicated().sum() #the total number of duplicate rows
    if total_duplicates == 0:
        print("No duplicate records exist.")
        return False
    print(f'A total of {total_duplicates} duplicates exist.')
    return True

def show_uniques(df: DataFrame, ignore:list[str]=[]):
    '''
    Prints the values in the columns of the given dataframe, ignoring any columns passed to the ignore argument.
    '''
    for column in df.columns:
        if column in ignore: #if a column has been specified in ignore
            continue #ignore the column
        counter = 0
        print(column)
        for item in df[column].unique():
            counter +=1
            if counter%3==0:
                print(item) #print an item and line break
                counter = 0
            else:
                print(item, end="\t\t") #print an item with spacing
        print("\n") #section break to new column

def show_details(df:DataFrame, column:str) -> Series:
    '''
    Returns statistics of sum, mean, standard deviation, skewness, and kurtosis of the given column as a Series.
    '''
    if type(df[column][0]) == str:
        print('Column does not contain numeric values.')
        return
    value = df[column]
    result = {
        'Sum':value.sum(),
        'Mean':value.mean(),
        'Standard Deviation': value.std(),
        'Skewness':value.skew(),
        'Kurtosis':value.kurt()
    }
    return Series(data=result, name=column)

def get_top_jobs(df: DataFrame, n:int = 15) -> Series:
    '''
    Gets the top jobs in the dataframe. By default it gets the top 15 jobs.
    '''
    total_count = df.groupby(by="job_title")["job_title"].count() #gets the total count of jobs
    top_jobs = total_count.sort_values(ascending=False) #sorts the values in descending order
    top_n_jobs = top_jobs.head(n) #gets only the top n jobs
    return top_n_jobs

def salary_experience(df:DataFrame) -> DataFrame:
    '''
    Gets the highest, lowest, and average salary based on experience, returning a dataframe.
    '''
    salary_grouped = df.groupby(by="experience_level")['salary_in_usd']
    mean_salary = salary_grouped.mean()
    mean_salary.name = 'Mean'
    max_salary = salary_grouped.max()
    max_salary.name = 'Max'
    min_salary = salary_grouped.min()
    min_salary.name = 'Min'
    return pd.concat([mean_salary,max_salary,min_salary],axis=1)
